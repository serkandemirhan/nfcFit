import { recordedSetProgress } from "./lib/trainer-planning";
import TrainerPlanning from "./TrainerPlanning";
import { consoleTransport } from "../mobile/nfc-task-tracker/lib/console-transport";
import {
  exercisePerformance,
  zonedDate,
  lastFinishedMonday,
} from "../mobile/nfc-task-tracker/lib/coaching-performance";
import React, { useEffect, useRef, useState } from "react";
import type {
  CoachingOverview,
  Exercise,
} from "../mobile/nfc-task-tracker/lib/coaching-types";
const input =
  "min-h-12 w-full rounded-xl border border-slate-600 bg-slate-900 px-3 py-2 text-white";
const action =
  "min-h-12 rounded-xl bg-lime-300 px-4 py-2 font-bold text-slate-950 disabled:opacity-40";
const secondary =
  "min-h-12 rounded-xl border border-slate-600 px-4 py-2 text-slate-100";
const card = "space-y-4 rounded-2xl border border-slate-700 bg-slate-900 p-5";
const date = (d = new Date()) =>
  `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
function Field({
  label,
  children,
}: {
  label: string;
  children: React.ReactNode;
}) {
  const id = React.useId();
  return (
    <div className="block space-y-2 text-sm text-slate-300">
      <label htmlFor={id}>{label}</label>
      {React.cloneElement(children as React.ReactElement<{ id?: string }>, {
        id,
      })}
    </div>
  );
}
type Summary = {
  id: string;
  athlete_id: string;
  tenant_id: string;
  name: string;
  active_plans: number;
  completion_rate: number | null;
  missed: number;
  unreviewed_changes: number;
  unread_messages: number;
  pending_reports: number;
  recent_workouts?: import("../mobile/nfc-task-tracker/lib/coaching-types").Workout[];
};
export default function CoachingConsole({
  token,
  tenantId,
}: {
  token: string;
  tenantId: string;
}) {
  const generation = useRef(0);
  const messageRequest = useRef<{ id: string; fingerprint: string } | null>(
    null,
  );
  const [data, setData] = useState<CoachingOverview | null>(null);
  const [catalog, setCatalog] = useState<Exercise[]>([]);
  const [summaries, setSummaries] = useState<Summary[]>([]);
  const [athlete, setAthlete] = useState("");
  const [tab, setTab] = useState("dashboard");
  const [createSignal, setCreateSignal] = useState(0);
  const [requestedPlanId, setRequestedPlanId] = useState("");
  const [editorFocused, setEditorFocused] = useState(false);
  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");
  const [busy, setBusy] = useState(false);
  const [search, setSearch] = useState("");
  const [attention, setAttention] = useState(false);
  const [reviewFilter, setReviewFilter] = useState("");
  const [from, setFrom] = useState("");
  const [until, setUntil] = useState("");
  const [planFilter, setPlanFilter] = useState("");
  const [stateFilter, setStateFilter] = useState("");
  const [reportFilter, setReportFilter] = useState("");
  const [exerciseFilter, setExerciseFilter] = useState("");
  const [body, setBody] = useState("");
  const [context, setContext] = useState<{ type: string; id: string } | null>(
    null,
  );
  const [week, setWeek] = useState(() => {
    const d = new Date();
    d.setDate(d.getDate() - ((d.getDay() + 6) % 7) - 7);
    return date(d);
  });
  useEffect(() => {
    if (data?.timezone) setWeek(lastFinishedMonday(data.timezone));
  }, [data?.timezone, athlete]);
  const api = async <T,>(
    path: string,
    payload?: unknown,
    method?: string,
  ): Promise<T> => {
    return consoleTransport(
      `${(import.meta.env.VITE_CONSOLE_API_URL ?? "").replace(/\/+$/, "")}/api/console/coaching${path}`,
      {
        method: method ?? (payload ? "POST" : "GET"),
        headers: {
          Authorization: `Bearer ${token}`,
          ...(payload ? { "Content-Type": "application/json" } : {}),
        },
        body: payload ? JSON.stringify(payload) : undefined,
      },
      async (response) => {
        if (!response.ok) {
          const value = await response.json().catch(() => null);
          throw new Error(
            typeof value?.detail === "string"
              ? value.detail
              : "İşlem tamamlanamadı. Alanları kontrol edip yeniden deneyin.",
          );
        }
        return await response.json();
      },
    );
  };

  const refresh = async () => {
    const epoch = generation.current;
    const [overview, rows, exercises] = await Promise.all([
      api<CoachingOverview>(
        "/overview" + (athlete ? "?athlete_id=" + athlete : ""),
      ),
      api<Summary[]>("/dashboard"),
      api<Exercise[]>("/library"),
    ]);
    if (epoch !== generation.current) return;
    setData(overview);
    setSummaries(rows);
    setCatalog(exercises);
  };
  useEffect(() => {
    let live = true;
    generation.current++;
    setData(null);
    setBusy(true);
    setError("");
    Promise.all([
      api<CoachingOverview>(
        "/overview" + (athlete ? "?athlete_id=" + athlete : ""),
      ),
      api<Summary[]>("/dashboard"),
      api<Exercise[]>("/library"),
    ])
      .then(([v, s, c]) => {
        if (live) {
          setData(v);
          setSummaries(s);
          setCatalog(c);
        }
      })
      .catch((e) => {
        if (live) setError(e.message);
      })
      .finally(() => {
        if (live) setBusy(false);
      });
    return () => {
      live = false;
    };
  }, [token, athlete, tenantId]); // All backend reads are scoped independently of these controls.
  const perform = async (
    fn: () => Promise<unknown>,
    message = "Değişiklik kaydedildi.",
  ) => {
    if (busy) return;
    const epoch = generation.current;
    setBusy(true);
    setError("");
    setNotice("");
    try {
      await fn();
      if (epoch !== generation.current) return;
      await refresh();
      setNotice(message);
    } catch (e) {
      if (epoch === generation.current) setError((e as Error).message);
    } finally {
      if (epoch === generation.current) setBusy(false);
    }
  };
  const filteredWorkouts = (data?.workouts ?? []).filter(
    (w) =>
      (!tenantId ||
        data?.plans.some(
          (p) => p.id === w.plan_id && p.tenant_id === tenantId,
        )) &&
      (!from || w.scheduled_date >= from) &&
      (!until || w.scheduled_date <= until) &&
      (!planFilter || w.plan_id === planFilter) &&
      (!stateFilter || w.state === stateFilter),
  );
  const performance = exercisePerformance(
    filteredWorkouts,
    data?.timezone ?? "Europe/Istanbul",
  );
  const selectedPerformance =
    performance.find((s) => s.id === exerciseFilter) ?? performance[0];
  const performanceDays = selectedPerformance?.days.slice(-14) ?? [];
  const usesVolume = performanceDays.some((d) => d.volume != null);
  const performanceMax = Math.max(
    1,
    ...performanceDays.map((d) => (usesVolume ? (d.volume ?? 0) : d.sets)),
  );
  const plans = (data?.plans ?? []).filter(
    (p) => !tenantId || p.tenant_id === tenantId,
  );
  const memberToday = zonedDate(
    new Date(),
    data?.timezone ?? "Europe/Istanbul",
  );
  const selectedMember = summaries.find(
    (a) => a.athlete_id === athlete && (!tenantId || a.tenant_id === tenantId),
  );
  const chosenLink = data?.links.find(
    (l) =>
      l.athlete_id === athlete &&
      l.trainer_id === data.account_id &&
      (!tenantId || l.tenant_id === tenantId),
  );
  return (
    <section
      className="trainer-area space-y-5"
      data-editor-open={editorFocused}
      aria-label="Antrenör çalışma alanı"
    >
      {error && (
        <p role="alert" className="rounded-xl bg-red-950 p-4 text-red-200">
          {error}
        </p>
      )}
      {notice && (
        <p role="status" className="text-lime-200">
          {notice}
        </p>
      )}
      {busy && (
        <p role="status" className="text-slate-400">
          Yükleniyor…
        </p>
      )}
      <div className="trainer-navigation-row">
        <nav className="trainer-navigation" aria-label="Antrenör bölümleri">
          {[
            ["dashboard", "Genel bakış"],
            ["athletes", "Üyelerim"],
            ["plans", "Programlar"],
            ["workouts", "Antrenmanlar"],
            ["reports", "İlerleme"],
            ["messages", "Mesajlar"],
          ].map(([key, label]) => (
            <button
              key={key}
              aria-current={tab === key ? "page" : undefined}
              className={tab === key ? action : secondary}
              onClick={() => {
                if (
                  window.dispatchEvent(
                    new Event("trainer-leave", { cancelable: true }),
                  )
                ) {
                  if (key === "dashboard") setAthlete("");
                  setTab(key);
                }
              }}
            >
              {label}
            </button>
          ))}
        </nav>
        <button
          className={secondary + " trainer-refresh"}
          disabled={busy}
          onClick={() => void perform(refresh, "Veriler yenilendi.")}
        >
          Yenile
        </button>
      </div>
      {tab !== "plans" && tab !== "workouts" && tab !== "dashboard" && (
        <Field label="Üye seç">
          <select
            className={input}
            value={athlete}
            onChange={(e) => {
              setAthlete(e.target.value);
              setContext(null);
              setBody("");
            }}
          >
            <option value="">Tüm üyeler</option>
            {summaries
              .filter((a) => !tenantId || a.tenant_id === tenantId)
              .map((a) => (
                <option key={a.id} value={a.athlete_id}>
                  {a.name}
                </option>
              ))}
          </select>
        </Field>
      )}
      {(tab === "dashboard" || tab === "athletes" || tab === "reports") && (
        <>
          {tab === "dashboard" && (
            <>
              <div className="trainer-overview-hero">
                <div>
                  <h3>Üyelerin için bir sonraki adım.</h3>
                  <p>
                    Planları takip et, ihtiyaçları gör ve gelecek haftayı
                    hazırla.
                  </p>
                </div>
                <button
                  className={action}
                  onClick={() => {
                    setAthlete("");
                    setTab("plans");
                    setCreateSignal((n) => n + 1);
                  }}
                >
                  ＋ Program oluştur
                </button>
              </div>
              <div className="trainer-kpis">
                {[
                  [
                    "Atanmış üye",
                    summaries.filter((a) => a.tenant_id === tenantId).length,
                    () => {
                      setTab("athletes");
                      setAttention(false);
                    },
                  ],
                  [
                    "Programı olan",
                    summaries.filter(
                      (a) => a.tenant_id === tenantId && a.active_plans > 0,
                    ).length,
                    () => setTab("athletes"),
                  ],
                  [
                    "Program bekleyen",
                    summaries.filter(
                      (a) => a.tenant_id === tenantId && !a.active_plans,
                    ).length,
                    () => {
                      setTab("athletes");
                      setReviewFilter("needs_program");
                    },
                  ],
                  [
                    "İnceleme bekleyen",
                    summaries
                      .filter((a) => a.tenant_id === tenantId)
                      .reduce(
                        (n, a) => n + a.unreviewed_changes + a.pending_reports,
                        0,
                      ),
                    () => {
                      setTab("athletes");
                      setAttention(true);
                    },
                  ],
                ].map(([label, value, click]) => (
                  <button key={String(label)} onClick={click as () => void}>
                    <strong>{value as number}</strong>
                    <span>{label as string} ↗</span>
                  </button>
                ))}
              </div>
              <div className="grid gap-4 md:grid-cols-2">
                <article className={card}>
                  <h3 className="text-lg font-bold">
                    Son tamamlanan antrenmanlar
                  </h3>
                  {summaries
                    .filter((a) => a.tenant_id === tenantId)
                    .flatMap((a) =>
                      (a.recent_workouts ?? [])
                        .filter(
                          (w) =>
                            w.state === "completed" || w.state === "partial",
                        )
                        .map((w) => (
                          <button
                            key={w.id}
                            className={secondary + " text-left"}
                            onClick={() => {
                              setAthlete(a.athlete_id);
                              setTab("athletes");
                            }}
                          >
                            {a.name} · {w.snapshot.name}
                            <span className="block text-xs text-slate-400">
                              {w.scheduled_date} ·{" "}
                              {
                                w.actual.sets.filter(
                                  (s) => s.status === "completed",
                                ).length
                              }{" "}
                              kaydedilmiş set ↗
                            </span>
                          </button>
                        )),
                    )}
                  {!summaries.some(
                    (a) =>
                      a.tenant_id === tenantId &&
                      a.recent_workouts?.some(
                        (w) => w.state === "completed" || w.state === "partial",
                      ),
                  ) && (
                    <p className="text-slate-400">
                      Henüz tamamlanmış antrenman yok.
                    </p>
                  )}
                </article>
                <article className={card}>
                  <h3 className="text-lg font-bold">Program kütüphanesi</h3>
                  {plans
                    .filter(
                      (p) =>
                        !p.athlete_id &&
                        (p.content.planning_status ?? "published") !==
                          "archived",
                    )
                    .slice(0, 3)
                    .map((p) => (
                      <button
                        className={secondary + " text-left"}
                        key={p.id}
                        onClick={() => setTab("plans")}
                      >
                        {p.content.name} · {p.content.weeks} hafta ↗
                      </button>
                    ))}
                  <button
                    className={secondary}
                    onClick={() => {
                      setTab("athletes");
                      setAttention(true);
                    }}
                  >
                    İnceleme gerektiren üyeleri aç ↗
                  </button>
                </article>
              </div>
              <h3 className="text-xl font-bold">Üye takibi</h3>
            </>
          )}
          {!athlete && (
            <>
              <div className="flex flex-wrap gap-3">
                <input
                  aria-label="Sporcu ara"
                  className={input + " sm:max-w-sm"}
                  placeholder="Sporcu ara"
                  value={search}
                  onChange={(e) => setSearch(e.target.value)}
                />
                <button
                  className={secondary}
                  onClick={() => setAttention(!attention)}
                >
                  {attention ? "Tüm sporcular" : "İlgi bekleyenler"}
                </button>
              </div>
              <Field label="İnceleme filtresi">
                <select
                  className={input}
                  value={reviewFilter}
                  onChange={(e) => setReviewFilter(e.target.value)}
                >
                  <option value="">Tüm durumlar</option>
                  <option value="needs_program">Program bekleyen</option>
                  <option value="missed">Kaçırılan seans</option>
                  <option value="unreviewed_changes">
                    İncelenmemiş değişiklik
                  </option>
                  <option value="unread_messages">Okunmamış mesaj</option>
                  <option value="pending_reports">Bekleyen rapor</option>
                </select>
              </Field>
              <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
                {summaries
                  .filter(
                    (a) =>
                      (!tenantId || a.tenant_id === tenantId) &&
                      a.name
                        .toLocaleLowerCase("tr")
                        .includes(search.toLocaleLowerCase("tr")) &&
                      (!reviewFilter ||
                        (reviewFilter === "needs_program"
                          ? !a.active_plans
                          : Number(a[reviewFilter as keyof Summary]) > 0)) &&
                      (!attention ||
                        a.missed +
                          a.unreviewed_changes +
                          a.unread_messages +
                          a.pending_reports >
                          0),
                  )
                  .map((a) => (
                    <article key={a.id} className={card}>
                      <h3 className="text-xl font-bold">{a.name}</h3>
                      <p>
                        {a.active_plans} program · Haftalık tamamlama{" "}
                        {a.completion_rate == null
                          ? "—"
                          : "%" + Math.round(a.completion_rate * 100)}
                      </p>
                      <p className="text-slate-400">
                        {a.missed} kaçırılan · {a.unreviewed_changes} değişiklik
                        · {a.unread_messages} mesaj · {a.pending_reports} rapor
                      </p>
                      <button
                        className={action}
                        onClick={() => {
                          setAthlete(a.athlete_id);
                          setTab("athletes");
                        }}
                      >
                        Sporcuyu incele
                      </button>
                    </article>
                  ))}
              </div>
              {!summaries.length && (
                <p className="text-slate-400">
                  Yalnızca sana açıkça atanmış aktif sporcular gösterilir. Tesis
                  yöneticisi Yönetim → Personel bölümünden atama yapabilir.
                </p>
              )}
            </>
          )}
          {athlete && (
            <>
              <header className="trainer-member-header">
                <button
                  className={secondary}
                  onClick={() => {
                    setAthlete("");
                    setFrom("");
                    setUntil("");
                    setPlanFilter("");
                    setStateFilter("");
                  }}
                >
                  ← Üyelere dön
                </button>
                <div>
                  <p className="text-xs text-lime-300">ÜYE DETAYI</p>
                  <h2>{selectedMember?.name}</h2>
                </div>
              </header>
              <div
                className="trainer-member-stats"
                aria-label="Üye antrenman özeti"
              >
                <div>
                  <strong>{selectedMember?.active_plans ?? 0}</strong>
                  <span>Aktif program</span>
                </div>
                <div>
                  <strong>
                    {
                      filteredWorkouts.filter((w) => w.state === "completed")
                        .length
                    }
                  </strong>
                  <span>Tamamlanan seans</span>
                </div>
                <div>
                  <strong>{selectedMember?.missed ?? 0}</strong>
                  <span>Kaçırılan seans</span>
                </div>
              </div>
              <section
                className="trainer-week-summary"
                aria-label="Haftalık ilerleme ve son antrenmanlar"
              >
                <div>
                  <h3>Haftalık tamamlama</h3>
                  {selectedMember?.completion_rate == null ? (
                    <p>Bu hafta için planlanmış seans yok.</p>
                  ) : (
                    <>
                      <div
                        className="trainer-week-progress"
                        role="progressbar"
                        aria-label="Haftalık tamamlanan antrenman oranı"
                        aria-valuemin={0}
                        aria-valuemax={100}
                        aria-valuenow={Math.round(
                          selectedMember.completion_rate * 100,
                        )}
                      >
                        <span
                          style={{
                            width:
                              Math.min(
                                100,
                                selectedMember.completion_rate * 100,
                              ) + "%",
                          }}
                        />
                      </div>
                      <p>
                        %{Math.round(selectedMember.completion_rate * 100)}{" "}
                        tamamlandı · {selectedMember?.missed ?? 0} kaçırılan
                        seans
                      </p>
                    </>
                  )}
                  <small>
                    Program hedefleri ve kaydedilmiş seanslara göre.
                  </small>
                </div>
                <div className="trainer-recent-history">
                  <h3>Son antrenmanlar</h3>
                  {filteredWorkouts.length ? (
                    [...filteredWorkouts]
                      .sort((a, b) =>
                        b.scheduled_date.localeCompare(a.scheduled_date),
                      )
                      .slice(0, 3)
                      .map((w) => (
                        <a href={"#workout-" + w.id} key={w.id}>
                          <span>
                            {w.scheduled_date} · {w.snapshot.name}
                          </span>
                          <strong>
                            {
                              w.actual.sets.filter(
                                (s) => s.status === "completed",
                              ).length
                            }{" "}
                            set
                          </strong>
                        </a>
                      ))
                  ) : (
                    <p>
                      {data?.workouts.length
                        ? "Bu filtrelerde antrenman kaydı yok. Filtreleri temizleyerek geçmişi görebilirsin."
                        : "Henüz antrenman kaydı yok. İlk seans kaydedildiğinde burada görünür."}
                    </p>
                  )}
                  <small>
                    Geçmiş hedefler tamamlanan antrenmanın kayıt anındaki
                    kopyasından gösterilir.
                  </small>
                </div>
              </section>
              <div className={card}>
                <h3 className="text-xl font-bold">Antrenman programları</h3>
                {plans
                  .filter(
                    (p) =>
                      p.athlete_id === athlete &&
                      p.content.planning_status !== "archived" &&
                      p.content.start_date <= memberToday,
                  )
                  .map((p) => (
                    <div
                      key={p.id}
                      className="flex flex-wrap items-center justify-between gap-3"
                    >
                      <span>
                        {p.content.name} · {p.content.weeks} hafta ·{" "}
                        {Math.max(
                          1,
                          Math.min(
                            p.content.weeks,
                            Math.floor(
                              (Date.now() - Date.parse(p.content.start_date)) /
                                604800000,
                            ) + 1,
                          ),
                        )}
                        . hafta
                      </span>
                      <button
                        className={secondary}
                        onClick={() => {
                          setRequestedPlanId(p.id);
                          setTab("plans");
                        }}
                      >
                        Gelecek planı düzenle
                      </button>
                    </div>
                  ))}
                {plans.some(
                  (p) =>
                    p.athlete_id === athlete &&
                    p.content.planning_status !== "archived" &&
                    p.content.start_date > memberToday,
                ) && (
                  <details className="trainer-scheduled-programs">
                    <summary>
                      Gelecek programlar (
                      {
                        plans.filter(
                          (p) =>
                            p.athlete_id === athlete &&
                            p.content.planning_status !== "archived" &&
                            p.content.start_date > memberToday,
                        ).length
                      }
                      )
                    </summary>
                    {plans
                      .filter(
                        (p) =>
                          p.athlete_id === athlete &&
                          p.content.planning_status !== "archived" &&
                          p.content.start_date > memberToday,
                      )
                      .map((p) => (
                        <div key={p.id} className="trainer-scheduled-row">
                          <span>
                            {p.content.name}
                            <small>
                              {p.content.start_date} · {p.content.weeks} hafta ·
                              Planlanan başlangıç
                            </small>
                          </span>
                          <button
                            className={secondary}
                            onClick={() => {
                              setRequestedPlanId(p.id);
                              setTab("plans");
                            }}
                          >
                            Gelecek planı düzenle
                          </button>
                        </div>
                      ))}
                  </details>
                )}
                {!plans.some(
                  (p) =>
                    p.athlete_id === athlete &&
                    p.content.planning_status !== "archived" &&
                    p.content.start_date <= memberToday,
                ) && (
                  <p className="text-slate-400">
                    Bu üyede henüz başlamış program bulunmuyor.
                  </p>
                )}
              </div>
              <details className={card}>
                <summary className="cursor-pointer font-bold">
                  Antrenman ve gelişim filtreleri
                </summary>
                <div className="grid gap-3 sm:grid-cols-2">
                  <Field label="Başlangıç">
                    <input
                      className={input}
                      type="date"
                      value={from}
                      onChange={(e) => setFrom(e.target.value)}
                    />
                  </Field>
                  <Field label="Bitiş">
                    <input
                      className={input}
                      type="date"
                      value={until}
                      onChange={(e) => setUntil(e.target.value)}
                    />
                  </Field>
                  <Field label="Program">
                    <select
                      className={input}
                      value={planFilter}
                      onChange={(e) => setPlanFilter(e.target.value)}
                    >
                      <option value="">Tüm programlar</option>
                      {data?.plans
                        .filter((p) => p.tenant_id === tenantId)
                        .map((p) => (
                          <option key={p.id} value={p.id}>
                            {p.content.name}
                          </option>
                        ))}
                    </select>
                  </Field>
                  <Field label="Antrenman durumu">
                    <select
                      className={input}
                      value={stateFilter}
                      onChange={(e) => setStateFilter(e.target.value)}
                    >
                      <option value="">Tüm durumlar</option>
                      {Object.entries({
                        not_started: "Başlanmadı",
                        in_progress: "Devam ediyor",
                        paused: "Duraklatıldı",
                        completed: "Tamamlandı",
                        partial: "Kısmen tamamlandı",
                        cancelled: "İptal edildi",
                      }).map(([key, label]) => (
                        <option key={key} value={key}>
                          {label}
                        </option>
                      ))}
                    </select>
                  </Field>
                </div>
                <button
                  className={secondary}
                  onClick={() => {
                    setFrom("");
                    setUntil("");
                    setPlanFilter("");
                    setStateFilter("");
                  }}
                >
                  Filtreleri temizle
                </button>
              </details>
              <details className={card}>
                <summary className="cursor-pointer font-bold">
                  Egzersiz gelişimi ve ayrıntılı analiz
                </summary>
                {!selectedPerformance ? (
                  <p>Karşılaştırma için henüz gerçekleşen set yok.</p>
                ) : (
                  <>
                    <Field label="Egzersiz">
                      <select
                        className={input}
                        value={selectedPerformance.id}
                        onChange={(e) => setExerciseFilter(e.target.value)}
                      >
                        {performance.map((p) => (
                          <option key={p.id} value={p.id}>
                            {p.name}
                          </option>
                        ))}
                      </select>
                    </Field>
                    <p>
                      {usesVolume
                        ? "Kayıtlı hacim (kg × tekrar)"
                        : "Kaydedilen set sayısı"}{" "}
                      · {data?.timezone ?? "Europe/Istanbul"}
                    </p>
                    <div
                      className="flex h-48 items-end gap-2"
                      role="img"
                      aria-label="Son 14 kayıtlı günün gelişimi. Değerler aşağıdaki tabloda yer alır."
                    >
                      {performanceDays.map((d) => (
                        <div
                          key={d.date}
                          className="flex min-w-0 flex-1 flex-col items-center gap-2"
                        >
                          <span className="text-xs">
                            {usesVolume ? (d.volume ?? "—") : d.sets}
                          </span>
                          <div
                            className="w-full rounded-t bg-lime-300"
                            style={{
                              height: Math.max(
                                3,
                                ((usesVolume ? (d.volume ?? 0) : d.sets) /
                                  performanceMax) *
                                  120,
                              ),
                            }}
                          />
                          <span className="text-xs">{d.date.slice(5)}</span>
                        </div>
                      ))}
                    </div>
                    <p className="text-sm text-slate-400">
                      Gerçek set tarihine göre gruplanır. Eksik ağırlıklar
                      tahmin edilmez. Yukarıdaki tarihler planlanan seans
                      tarihini filtreler.
                    </p>
                    <div className="overflow-x-auto">
                      <table className="w-full text-left text-sm">
                        <thead>
                          <tr>
                            <th>Tarih</th>
                            <th>Set</th>
                            <th>Tekrar</th>
                            <th>En yüksek kg</th>
                            <th>Hacim</th>
                            <th>Süre (sn)</th>
                            <th>Mesafe (m)</th>
                          </tr>
                        </thead>
                        <tbody>
                          {performanceDays.map((d) => (
                            <tr key={d.date}>
                              <td>{d.date}</td>
                              <td>{d.sets}</td>
                              <td>{d.reps ?? "—"}</td>
                              <td>{d.maxKg ?? "—"}</td>
                              <td>{d.volume ?? "—"}</td>
                              <td>{d.seconds ?? "—"}</td>
                              <td>{d.meters ?? "—"}</td>
                            </tr>
                          ))}
                        </tbody>
                      </table>
                    </div>
                  </>
                )}
              </details>
            </>
          )}
          {athlete &&
            filteredWorkouts.map((w) => (
              <article
                className={card + " trainer-workout-progress"}
                id={"workout-" + w.id}
                key={w.id}
                data-workout-id={w.id}
              >
                <h3 className="font-bold">
                  {w.scheduled_date} · {w.snapshot.name}
                </h3>
                <p>
                  {
                    {
                      not_started: "Başlanmadı",
                      in_progress: "Devam ediyor",
                      paused: "Duraklatıldı",
                      completed: "Tamamlandı",
                      partial: "Kısmen tamamlandı",
                      skipped: "Atlandı",
                      cancelled: "İptal edildi",
                    }[w.state]
                  }
                </p>
                {[...w.snapshot.exercises, ...w.actual.added].map((e) => {
                  const progress = recordedSetProgress(w, e.id);
                  return (
                    <section
                      key={e.id}
                      className="trainer-exercise-progress"
                      data-extra={progress.extra}
                      data-completed={progress.completed}
                      data-planned={progress.planned}
                    >
                      <h4>
                        {e.name ??
                          catalog.find((x) => x.id === e.exercise_id)?.name ??
                          e.exercise_id}
                      </h4>
                      <dl className="trainer-set-counts">
                        <div>
                          <dt>Planlanan set</dt>
                          <dd>{progress.planned}</dd>
                        </div>
                        <div>
                          <dt>Tamamlanan set</dt>
                          <dd>{progress.completed}</dd>
                        </div>
                        <div className={progress.extra ? "has-extra" : ""}>
                          <dt>Ekstra set</dt>
                          <dd>{progress.extra ? `+${progress.extra}` : 0}</dd>
                        </div>
                      </dl>
                      <div
                        className="trainer-set-chart"
                        role="img"
                        aria-label={`${progress.planned} planlanan, ${progress.completed} tamamlanan, ${progress.extra} ekstra set`}
                      >
                        <div className="trainer-set-track">
                          <span
                            style={{
                              width: progress.planned
                                ? Math.min(
                                    100,
                                    (progress.completed / progress.planned) *
                                      100,
                                  ) + "%"
                                : "0%",
                            }}
                          />
                        </div>
                        <small>
                          {progress.planned
                            ? "Planlanan hedef"
                            : "Eklenen hareket"}
                          {progress.extra ? ` · +${progress.extra} ekstra` : ""}
                        </small>
                      </div>
                      <p className="trainer-set-status">
                        {progress.completed}/{progress.planned} ·{" "}
                        {progress.extra ? `+${progress.extra} ekstra · ` : ""}
                        {progress.achieved
                          ? "Hedefe ulaşıldı"
                          : progress.planned
                            ? "Hedef sürüyor"
                            : "Plansız kayıt"}
                      </p>
                      <details>
                        <summary className="cursor-pointer py-3 text-sm">
                          Kaydedilen setleri göster
                        </summary>
                        {w.actual.sets
                          .filter((s) => s.exercise_key === e.id)
                          .map((s) => (
                            <p key={s.id}>
                              Set {s.set_number}: {s.reps ?? "—"} tekrar ·{" "}
                              {s.kg ?? "—"} kg · {s.seconds ?? "—"} sn ·{" "}
                              {s.meters ?? "—"} m ·{" "}
                              {s.status === "completed"
                                ? "Tamamlandı"
                                : "Atlandı"}{" "}
                              ·{" "}
                              {s.method === "nfc"
                                ? "NFC ile kayıt"
                                : "Manuel giriş"}{" "}
                              · {s.notes}
                            </p>
                          ))}
                      </details>
                    </section>
                  );
                })}
                <p>{w.actual.notes}</p>
                <button
                  className={secondary}
                  onClick={() => {
                    setContext({ type: "workout", id: w.id });
                    setTab("messages");
                  }}
                >
                  Antrenmana yorum yaz
                </button>
              </article>
            ))}
        </>
      )}
      {(tab === "plans" || tab === "workouts") && (
        <TrainerPlanning
          key={tenantId}
          api={api}
          plans={plans}
          catalog={catalog}
          members={summaries}
          tenantId={tenantId}
          timezone={data?.timezone ?? "Europe/Istanbul"}
          mode={tab}
          comment={(revisionId) => {
            setContext({ type: "revision", id: revisionId });
            setTab("messages");
          }}
          reload={refresh}
          createSignal={createSignal}
          onEditorFocusChange={setEditorFocused}
          openPlanId={requestedPlanId}
          onPlanOpened={() => setRequestedPlanId("")}
        />
      )}
      {tab === "reports" && (
        <>
          <div className={card}>
            <Field label="Bitmiş hafta başlangıcı (Pazartesi)">
              <input
                className={input}
                type="date"
                value={week}
                onChange={(e) => setWeek(e.target.value)}
              />
            </Field>
            <button
              className={action}
              disabled={!athlete || busy}
              onClick={() =>
                void perform(() =>
                  api(`/athletes/${athlete}/reports`, {
                    week,
                    timezone: data?.timezone ?? "Europe/Istanbul",
                    tenant_id: tenantId,
                  }),
                )
              }
            >
              Rapor oluştur / güncelle
            </button>
          </div>
          <Field label="Rapor durumu">
            <select
              className={input}
              value={reportFilter}
              onChange={(e) => setReportFilter(e.target.value)}
            >
              <option value="">Tüm raporlar</option>
              {Object.entries({
                awaiting_review: "İnceleme bekliyor",
                reviewed: "İncelendi",
                revision_recommended: "Revizyon önerildi",
                closed: "Kapatıldı",
              }).map(([key, label]) => (
                <option key={key} value={key}>
                  {label}
                </option>
              ))}
            </select>
          </Field>
          {data?.reports
            .filter(
              (r) =>
                (!tenantId || r.scope === tenantId) &&
                (!reportFilter || r.status === reportFilter),
            )
            .map((r) => (
              <article key={r.id} className={card}>
                <h3 className="text-xl font-bold">
                  {r.week} · {r.status}
                </h3>
                <p>
                  Seans {r.content.completed_sessions}/
                  {r.content.scheduled_sessions} · Kaçırılan{" "}
                  {r.content.missed_sessions} · Kısmi{" "}
                  {r.content.partial_sessions}
                </p>
                <p>
                  Set {r.content.completed_sets}/{r.content.planned_sets} ·
                  Tekrar {r.content.actual_reps}/{r.content.planned_reps}
                </p>
                <p>
                  Kayıtlı hacim:{" "}
                  {r.content.recorded_volume_kg_reps == null
                    ? "Yeterli veri yok"
                    : r.content.recorded_volume_kg_reps + " kg × tekrar"}
                </p>
                <p className="text-slate-400">
                  Tamamlama ile planlı set uyumu ayrı ölçülür. Eksik ağırlık
                  tahmin edilmez.
                </p>
                <div className="flex flex-wrap gap-3">
                  {["reviewed", "revision_recommended", "closed"].map(
                    (status, i) => (
                      <button
                        key={status}
                        className={secondary}
                        disabled={busy}
                        onClick={() =>
                          void perform(() =>
                            api(`/reports/${r.id}/review`, { status }),
                          )
                        }
                      >
                        {["İncelendi", "Revizyon öner", "Kapat"][i]}
                      </button>
                    ),
                  )}
                  <button
                    className={action}
                    onClick={() => {
                      setContext({ type: "report", id: r.id });
                      setTab("messages");
                    }}
                  >
                    Geri bildirim yaz
                  </button>
                </div>
              </article>
            ))}
        </>
      )}
      {tab === "messages" && (
        <article className={card}>
          <h3 className="font-bold">
            {chosenLink?.name ?? "Önce bir sporcu seç"}
          </h3>
          {context && (
            <p className="text-slate-400">
              Seçili kayda ilişkin geri bildirim{" "}
              <button onClick={() => setContext(null)} className={secondary}>
                Genel mesaja dön
              </button>
            </p>
          )}
          <textarea
            aria-label="Mesaj"
            className={input}
            value={body}
            onChange={(e) => setBody(e.target.value)}
            maxLength={4000}
          />
          <button
            className={action}
            disabled={!chosenLink || !body.trim() || busy}
            onClick={() =>
              void perform(async () => {
                const fingerprint = JSON.stringify([
                  chosenLink!.id,
                  body.trim(),
                  context,
                ]);
                if (messageRequest.current?.fingerprint !== fingerprint)
                  messageRequest.current = {
                    id: crypto.randomUUID(),
                    fingerprint,
                  };
                await api("/messages", {
                  id: messageRequest.current.id,
                  link_id: chosenLink!.id,
                  body: body.trim(),
                  context_type: context?.type ?? "general",
                  context_id: context?.id ?? null,
                });
                messageRequest.current = null;
                setBody("");
              })
            }
          >
            Gönder
          </button>
          {data?.messages
            .filter(
              (m) =>
                m.link_id === chosenLink?.id &&
                (!context || m.context_id === context.id),
            )
            .map((m) => (
              <div
                key={m.id}
                className="space-y-2 border-t border-slate-700 pt-4"
              >
                <p className="text-sm text-slate-400">
                  {m.author_id === data.account_id ? "Sen" : chosenLink?.name} ·{" "}
                  {new Date(m.created_at).toLocaleString("tr-TR")} ·{" "}
                  {m.read_at ? "Okundu" : "Okunmadı"}
                </p>
                <p>{m.body}</p>
                {m.author_id !== data.account_id && !m.read_at && (
                  <button
                    className={secondary}
                    onClick={() =>
                      void perform(() => api(`/messages/${m.id}/read`, {}))
                    }
                  >
                    Okundu
                  </button>
                )}
              </div>
            ))}
        </article>
      )}
    </section>
  );
}
